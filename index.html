    // ==============================================================================
// Yukti AI - ESP32 Field Telemetry Transmitter
// Board: ESP32 Dev Module | Arduino IDE 2.x
// ==============================================================================
#include <WiFi.h>
#include <HTTPClient.h>
#include <ArduinoJson.h>

const char* ssid = "YOUR_WIFI_SSID";
const char* password = "YOUR_WIFI_PASS";

// Point to your Hugging Face Space URL or local IP:
// Example: "https://your-user-yukti-ai.hf.space/api/iot/feed"
const char* serverUrl = "https://your-space.hf.space/api/iot/feed";

const int WATER_SENSOR_PIN = 34; // Analog depth probe
const int RELAY_PUMP_PIN = 23;   // Pump motor relay

void setup() {
  Serial.begin(115200);
  pinMode(RELAY_PUMP_PIN, OUTPUT);
  digitalWrite(RELAY_PUMP_PIN, LOW);

  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }
  Serial.println("\nWiFi Connected! IP: " + WiFi.localIP().toString());
}

void loop() {
  if (WiFi.status() == WL_CONNECTED) {
    HTTPClient http;
    http.begin(serverUrl);
    http.addHeader("Content-Type", "application/json");

    int rawAdc = analogRead(WATER_SENSOR_PIN);
    float waterLevelCm = (rawAdc / 4095.0) * 35.0; // 0-35 cm
    float soilMoisturePct = 62.0;

    StaticJsonDocument<256> doc;
    doc["device_id"] = "ESP32-NODE-FIELD-01";
    doc["field_id"] = "FIELD_01";
    doc["water_level_cm"] = waterLevelCm;
    doc["soil_moisture_pct"] = soilMoisturePct;
    doc["rainfall_mm_per_hr"] = 1.2;

    String jsonPayload;
    serializeJson(doc, jsonPayload);

    int httpCode = http.POST(jsonPayload);
    if (httpCode > 0) {
      String response = http.getString();
      Serial.println("Response: " + response);

      StaticJsonDocument<512> resDoc;
      deserializeJson(resDoc, response);

      // Actuator directives directly from AI model
      const char* pumpCommand = resDoc["actuator_directives"]["pump_command"];
      if (strcmp(pumpCommand, "DISCHARGE_100_PERCENT") == 0 || strcmp(pumpCommand, "AUXILIARY_PUMP_1") == 0) {
        digitalWrite(RELAY_PUMP_PIN, HIGH); // Engage pump
        Serial.println("AI DIRECTIVE: Pump ENGAGED!");
      } else {
        digitalWrite(RELAY_PUMP_PIN, LOW);  // Disengage pump
      }
    }
    http.end();
  }

  // Transmit telemetry every 3 hours
  delay(10800000); // 3 hours in milliseconds
}

# Yukti AI - Python Edge Gateway (Raspberry Pi / Jetson / Field Station)

import time
import requests

SERVER_URL = "https://your-space.hf.space/api/iot/feed"

def send_field_telemetry():
    payload = {
        "device_id": "RPI-GATEWAY-SECTOR-4",
        "field_id": "FIELD_01",
        "water_level_cm": 15.2,
        "soil_moisture_pct": 68.0,
        "rainfall_mm_per_hr": 3.4,
        "cumulative_rain_24h_mm": 18.5,
        "temperature_c": 26.5,
        "humidity_pct": 74.0,
        "drainage_flow_rate_lps": 8.0,
        "river_distance_m": 90.0
    }

    try:
        response = requests.post(SERVER_URL, json=payload, timeout=10)
        if response.status_code == 200:
            data = response.json()
            directives = data.get("actuator_directives", {})
            print(f"[✓] Transmitted! Risk: {data['prediction']['risk_label']}")
            print(f"    Pump Directive: {directives.get('pump_command')}")
            print(f"    Sluice Position: {directives.get('sluice_position_pct')}%")
        else:
            print(f"[!] Server returned status {response.status_code}: {response.text}")
    except Exception as e:
        print(f"[X] Transmission failed: {e}")

if __name__ == "__main__":
    while True:
        send_field_telemetry()
        # Ingest every 3 hours (10800 seconds)
        time.sleep(3 * 3600)

// Yukti AI - ESP32-CAM Remote Image Uploader
// Captures snapshot and pushes multipart HTTP POST to /api/iot/image-feed
// ==============================================================================
#include "esp_camera.h"
#include <WiFi.h>
#include <HTTPClient.h>

const char* serverUrl = "https://your-space.hf.space/api/iot/image-feed";

void captureAndSendImage() {
  camera_fb_t* fb = esp_camera_fb_get();
  if (!fb) {
    Serial.println("Camera capture failed!");
    return;
  }

  HTTPClient http;
  http.begin(serverUrl);

  String boundary = "----WebKitFormBoundaryYuktiAI";
  http.addHeader("Content-Type", "multipart/form-data; boundary=" + boundary);

  String head = "--" + boundary + "\r\n" +
                "Content-Disposition: form-data; name=\"device_id\"\r\n\r\n" +
                "ESP32-CAM-01\r\n" +
                "--" + boundary + "\r\n" +
                "Content-Disposition: form-data; name=\"task\"\r\n\r\n" +
                "plant_health\r\n" +
                "--" + boundary + "\r\n" +
                "Content-Disposition: form-data; name=\"file\"; filename=\"photo.jpg\"\r\n" +
                "Content-Type: image/jpeg\r\n\r\n";
  String tail = "\r\n--" + boundary + "--\r\n";

  size_t totalLen = head.length() + fb->len + tail.length();
  uint8_t* buffer = (uint8_t*)malloc(totalLen);
  memcpy(buffer, head.c_str(), head.length());
  memcpy(buffer + head.length(), fb->buf, fb->len);
  memcpy(buffer + head.length() + fb->len, tail.c_str(), tail.length());

  int httpCode = http.POST(buffer, totalLen);
  if (httpCode > 0) {
    Serial.println("Upload success: " + http.getString());
  }
  free(buffer);
  esp_camera_fb_return(fb);
  http.end();
}
